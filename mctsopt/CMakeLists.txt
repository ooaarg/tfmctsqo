cmake_minimum_required(VERSION 3.16)
project(mctsopt VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(mctsopt_core STATIC mctsopt_core/src/search.c)
set_target_properties(mctsopt_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mctsopt_core PUBLIC mctsopt_core/include)
target_compile_options(mctsopt_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
target_link_libraries(mctsopt_core PUBLIC m)
